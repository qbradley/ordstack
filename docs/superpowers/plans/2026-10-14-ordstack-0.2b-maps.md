# ordstack 0.2b (maps) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add `SMap` (a growable, insertion-ordered hash map that holds its frame like `SVec`) and `SFrozenMap` (its frozen form), replacing the sorted-pairs workaround from benchmark feedback 2.

**Architecture:** One new module, `crates/ordstack/src/map.rs`. A map's block holds `usable(slots)` entries `Bucket { hash: u64, key, value }` in insertion order, followed by an index of `slots` `u32` slots (0 = empty, otherwise entry index + 1), probed linearly, at most 7/8 full. Growth extends the block in place (the entries never move) and rebuilds the index at its new offset from the stored hashes, so it runs no user code. All probing is safe code on slices; index values are used only through bounds-checked indexing, so a wrong index is a logic error, never undefined behaviour. 15 new `unsafe` blocks, each a variant of an existing `SVec` one.

**Tech Stack:** Rust 2024, MSRV 1.85, `std::hash::{BuildHasher, RandomState}`, `trybuild` (compile-fail), Miri (Stacked and Tree Borrows), `cargo fuzz`, the existing `scripts/check.sh`.

**Spec:** `docs/superpowers/specs/2026-10-14-ordstack-0.2-design.md` (section 3 "B maps" is this plan; section 4 is its verification), which extends `docs/superpowers/specs/2026-10-13-ordstack-design.md`.

use ordstack::{AllocErrorKind, SMap, Stack};
use std::cell::RefCell;
use std::hash::{BuildHasher, Hash, Hasher};
use std::panic::{AssertUnwindSafe, catch_unwind};
use std::rc::Rc;

/// Pushes its id to a shared log when dropped.
struct D(u32, Rc<RefCell<Vec<u32>>>);
impl Drop for D {
    fn drop(&mut self) {
        self.1.borrow_mut().push(self.0);
    }
}
fn log() -> Rc<RefCell<Vec<u32>>> {
    Rc::new(RefCell::new(Vec::new()))
}

/// Hashes every key to `key % 4`: long probe runs, many collisions.
#[derive(Clone, Copy, Default)]
struct Collide;
struct CollideHasher(u64);
impl Hasher for CollideHasher {
    fn finish(&self) -> u64 {
        self.0 % 4
    }
    fn write(&mut self, bytes: &[u8]) {
        for b in bytes {
            self.0 = self.0.wrapping_mul(31).wrapping_add(*b as u64);
        }
    }
}
impl BuildHasher for Collide {
    type Hasher = CollideHasher;
    fn build_hasher(&self) -> CollideHasher {
        CollideHasher(0)
    }
}

#[test]
fn growth_extends_the_block_in_place() {
    let mut st = Stack::with_reserve(1 << 20);
    let mut f = st.frame();
    let _kept = f.alloc(1u64); // 0..8
    let mut m: SMap<u64, u64> = SMap::new_in(&mut f);
    assert_eq!(m.capacity(), 0);
    for i in 0..7 {
        m.insert(i, i * 10);
    }
    assert_eq!(m.capacity(), 7);
    let first = m.get(&0).unwrap() as *const u64;
    // 7 entries and an 8-slot index.
    assert_eq!(m.above().stack().extent(), 8 + 7 * E64 + 8 * 4);
    m.insert(7, 70); // grows: 14 entries and a 16-slot index
    assert_eq!(m.capacity(), 14);

    for i in 8..1000 {
        m.insert(i, i * 10);
    }
    assert_eq!(m.get(&0).unwrap() as *const u64, first); // never moved
    assert!((0..1000).all(|i| m.get(&i) == Some(&(i * 10))));
    assert!(m.keys().copied().eq(0..1000));
}

}

/// Random inserts and removals with a colliding hasher, against a `Vec`
/// model of the insertion order.
#[test]
fn heavy_collisions_match_a_model() {
    let mut st = Stack::with_reserve(1 << 20);
    let mut f = st.frame();
    let mut m: SMap<u32, u32, Collide> = SMap::with_hasher_in(Collide, &mut f);
    let mut model: Vec<(u32, u32)> = Vec::new();
    let mut seed = 0x9E37_79B9u32;
    let mut rnd = move || {
        seed ^= seed << 13;
        seed ^= seed >> 17;
        seed ^= seed << 5;
        seed
    };
    let ops = if cfg!(miri) { 150 } else { 5000 };
    for step in 0..ops {
        let k = rnd() % 64;
        let pos = model.iter().position(|e| e.0 == k);
        match rnd() % 4 {
            0 | 1 => {
                let old = m.insert(k, step);
                match pos {
                    Some(p) => assert_eq!(old, Some(std::mem::replace(&mut model[p].1, step))),
                    None => {
                        assert_eq!(old, None);
                        model.push((k, step));
                    }
                }
            }
            2 => assert_eq!(m.swap_remove(&k), pos.map(|p| model.swap_remove(p).1)),
            _ => assert_eq!(m.shift_remove(&k), pos.map(|p| model.remove(p).1)),
        }
        assert!(m.iter().map(|(k, v)| (*k, *v)).eq(model.iter().copied()));
        assert!((0..64).all(|k| m.get(&k) == model.iter().find(|e| e.0 == k).map(|e| &e.1)));
    }
}

/// A key whose `Hash` or `Eq` panics on demand.
struct Touchy {
    k: u32,
    panic_hash: bool,
}
impl Hash for Touchy {
    fn hash<H: Hasher>(&self, h: &mut H) {
        assert!(!self.panic_hash, "hash panicked");
        self.k.hash(h);
    }
}
impl PartialEq for Touchy {
    fn eq(&self, o: &Touchy) -> bool {
        assert!(self.k != 99 && o.k != 99, "eq panicked");
        self.k == o.k
    }
}
impl Eq for Touchy {}

#[test]
fn a_panicking_hash_or_eq_leaves_the_map_unchanged() {
    let mut st = Stack::with_reserve(1 << 16);
    let mut f = st.frame();
    let mut m: SMap<Touchy, u32, Collide> = SMap::with_hasher_in(Collide, &mut f);
    for k in 0..6 {
        m.insert(
            Touchy {
                k,
                panic_hash: false,
            },
            k,
        );
    }
    let key = |k, panic_hash| Touchy { k, panic_hash };
    assert!(catch_unwind(AssertUnwindSafe(|| m.insert(key(6, true), 6))).is_err());
    // 99 collides with keys already present, so `eq` runs and panics.
    assert!(catch_unwind(AssertUnwindSafe(|| m.insert(key(99, false), 99))).is_err());
    assert!(catch_unwind(AssertUnwindSafe(|| m.swap_remove(&key(99, false)))).is_err());
    assert_eq!(m.len(), 6);
    assert!((0..6).all(|k| m.get(&key(k, false)) == Some(&k)));
    m.insert(key(6, false), 6);
    assert_eq!(m.get_index_of(&key(6, false)), Some(6));
}

```rust
//! [`SMap`] and [`SFrozenMap`], insertion-ordered hash maps, with their
//! entry and iterator types.
//!
//! One block holds the entries, in insertion order, followed by an index
//! table of `u32` slots (0 is empty, otherwise the entry's index plus one),
//! probed linearly and at most 7/8 full. Growth extends the block in place:
//! the entries stay where they are and the index is rebuilt at its new
//! offset from the stored hashes, so growing runs no user code.

use core::borrow::Borrow;
use core::fmt;
use core::hash::{BuildHasher, Hash};
use core::marker::PhantomData;
use core::mem::{align_of, size_of};
use core::ptr::NonNull;
use std::hash::RandomState;

use crate::debug;
use crate::error::{AllocError, AllocErrorKind, fail};
use crate::frame::Frame;
use crate::slice::drop_back_to_front;

const OVERFLOW: AllocError = AllocError::new(AllocErrorKind::CapacityOverflow);

/// The most entries a map can hold: a slot stores an entry index plus one
/// in a `u32`.
const MAX_LEN: usize = u32::MAX as usize - 1;

/// One entry. Never zero-size, and aligned to at least 4 (it holds a
/// `u64`), so the index that follows the entries is aligned.
struct Bucket<K, V> {
    hash: u64,
    key: K,
    value: V,
}

/// The entries an index of `slots` slots may hold: 7/8 of it, so at least
/// one slot is always empty and every probe ends.
fn usable(slots: usize) -> usize {
    slots - slots / 8
}

/// The smallest index for `n` entries: 0 for none, otherwise a power of two
/// of at least 8.
fn slots_for(n: usize) -> Result<usize, AllocError> {
    if n == 0 {
        return Ok(0);
    }
    if n > MAX_LEN {
        return Err(OVERFLOW);
    }
    let mut s: usize = 8;
    while usable(s) < n {
        s = s.checked_mul(2).ok_or(OVERFLOW)?;
    }
    Ok(s)
}

/// Bytes of a block with an index of `slots` slots: `usable(slots)` entries
/// followed by the index.
fn block_bytes<K, V>(slots: usize) -> Result<usize, AllocError> {
    usable(slots)
        .checked_mul(size_of::<Bucket<K, V>>())
        .and_then(|e| e.checked_add(slots.checked_mul(4)?))
        .ok_or(OVERFLOW)
}

#[cold]
fn corrupt() -> ! {
    panic!("ordstack: internal error: map index is inconsistent")
}

/// Probes `index` (non-empty, power-of-two length) from the home slot of
/// `hash`. Returns the slot and entry of the first entry for which
/// `is_match` holds, or the first empty slot.
fn probe(
    index: &[u32],
    hash: u64,
    mut is_match: impl FnMut(usize) -> bool,
) -> Result<(usize, usize), usize> {
    let mask = index.len() - 1;
    let mut s = hash as usize & mask;
    for _ in 0..index.len() {
        match index[s] {
            0 => return Err(s),
            v => {
                let e = v as usize - 1;
                if is_match(e) {
                    return Ok((s, e));
                }
            }
        }
        s = (s + 1) & mask;
    }
    corrupt()
}

/// The slot that refers to `entry`, whose stored hash is `hash`.
fn slot_of(index: &[u32], hash: u64, entry: usize) -> usize {
    match probe(index, hash, |e| e == entry) {
        Ok((s, _)) => s,
        Err(_) => corrupt(),
    }
}

/// Records `entry` with `hash` in the first empty slot of its probe run.
fn insert_slot(index: &mut [u32], hash: u64, entry: usize) {
    match probe(index, hash, |_| false) {
        Err(s) => index[s] = entry as u32 + 1,
        Ok(_) => corrupt(),
    }
}

/// Empties slot `hole` and moves later slots of the same probe runs back
/// (backward-shift deletion), so no run is broken by the new empty slot.
fn remove_slot(index: &mut [u32], mut hole: usize, hash_of: impl Fn(usize) -> u64) {
    let mask = index.len() - 1;
    let mut s = (hole + 1) & mask;
    loop {
        let v = index[s];
        if v == 0 {
            break;
        }
        let home = hash_of(v as usize - 1) as usize & mask;
        // The entry may move back to `hole` unless its home lies cyclically
        // in `(hole, s]`.
        if s.wrapping_sub(home) & mask >= s.wrapping_sub(hole) & mask {
            index[hole] = v;
            hole = s;
        }
        s = (s + 1) & mask;
    }
    index[hole] = 0;
}

/// Fills an all-zero `index` with the entries whose hashes are `hashes`.
fn fill_index(index: &mut [u32], hashes: impl Iterator<Item = u64>) {
    for (e, h) in hashes.enumerate() {
        insert_slot(index, h, e);
    }
}

/// The entry for `q`, if any.
fn lookup<K, V, Q>(buckets: &[Bucket<K, V>], index: &[u32], hash: u64, q: &Q) -> Option<usize>
where
    K: Borrow<Q>,
    Q: ?Sized + Eq,
{
    if buckets.is_empty() {
        return None;
    }
    probe(index, hash, |e| {
        buckets[e].hash == hash && buckets[e].key.borrow() == q
    })
    .ok()
    .map(|(_, e)| e)
}

/// A growable, insertion-ordered hash map on a stack.
///
/// Like [`SVec`](crate::SVec) it holds its frame exclusively, so its block is
/// the top block of the stack and grows in place; [`above`](Self::above)
/// gives a frame above it. Iteration follows insertion order, so output
/// never depends on the hasher. Entries are dropped back to front.
///
/// # Examples
/// ```
/// use ordstack::{SMap, Stack};
///
/// let mut st = Stack::with_reserve(1 << 16);
/// let mut f = st.frame();
/// let mut m: SMap<&str, u32> = SMap::new_in(&mut f);
/// for w in "a b a c b a".split(' ') {
///     match m.get_mut(w) {
///         Some(n) => *n += 1,
///         None => {
///             m.insert(w, 1);
///         }
///     }
/// }
/// assert_eq!(m.get("a"), Some(&3));
/// assert_eq!(m.keys().copied().collect::<Vec<_>>(), ["a", "b", "c"]);
/// ```
pub struct SMap<'v, 'f, K, V, S = RandomState> {
    frame: &'v mut Frame<'f>,
    /// Start of the block; dangling while nothing is allocated.
    ptr: NonNull<Bucket<K, V>>,
    len: usize,
    /// Index slots, after `usable(slots)` entries; 0 while nothing is
    /// allocated.
    slots: usize,
    hasher: S,
    _owns: PhantomData<Bucket<K, V>>,
}

impl<'v, 'f, K, V> SMap<'v, 'f, K, V, RandomState> {
    /// An empty map on top of `frame`, with a fresh [`RandomState`].
    /// Allocates nothing yet.
    pub fn new_in(frame: &'v mut Frame<'f>) -> Self {
        SMap::with_hasher_in(RandomState::new(), frame)
    }

    /// An empty map with room for at least `n` entries.
    ///
    /// # Panics
    /// If the stack is exhausted, the capacity overflows, or an entry needs
    /// alignment above 4096.
    pub fn with_capacity_in(n: usize, frame: &'v mut Frame<'f>) -> Self {
        SMap::with_capacity_and_hasher_in(n, RandomState::new(), frame)
    }
}

impl<'v, 'f, K, V, S> SMap<'v, 'f, K, V, S> {
    /// An empty map on top of `frame` that hashes with `hasher`. Allocates
    /// nothing yet.
    pub fn with_hasher_in(hasher: S, frame: &'v mut Frame<'f>) -> Self {
        SMap {
            frame,
            ptr: NonNull::dangling(),
            len: 0,
            slots: 0,
            hasher,
            _owns: PhantomData,
        }
    }

    /// An empty map with room for at least `n` entries that hashes with
    /// `hasher`.
    ///
    /// # Panics
    /// As [`with_capacity_in`](SMap::with_capacity_in).
    pub fn with_capacity_and_hasher_in(n: usize, hasher: S, frame: &'v mut Frame<'f>) -> Self {
        let mut m = SMap::with_hasher_in(hasher, frame);
        m.reserve(n);
        m
    }

    fn allocated(&self) -> bool {
        self.slots > 0
    }

    /// Bytes of the block. Only meaningful when allocated, and then it
    /// cannot overflow: the block exists.
    fn bytes(&self) -> usize {
        usable(self.slots) * size_of::<Bucket<K, V>>() + self.slots * 4
    }

    /// Offset of the end of this map: its block's end, or the frame's end
    /// while nothing is allocated.
    fn end_off(&self) -> usize {
        if self.allocated() {
            self.frame.st.offset_of(self.ptr.cast()) + self.bytes()
        } else {
            self.frame.end
        }
    }

    fn check(&self) {
        debug::check_live(
            self.ptr.as_ptr().cast(),
            if self.allocated() { self.bytes() } else { 0 },
        );
    }

    /// The entries and the index.
    fn parts(&self) -> (&[Bucket<K, V>], &[u32]) {
        self.check();
        // SAFETY: M: `len` initialised entries at the start of the block and
        // `slots` initialised slots after `usable(slots)` entries, disjoint
        // from them. With nothing allocated both lengths are 0 and the
        // pointers are dangling and aligned.
        unsafe {
            let p = self.ptr.as_ptr();
            (
                core::slice::from_raw_parts(p, self.len),
                core::slice::from_raw_parts(p.add(usable(self.slots)).cast(), self.slots),
            )
        }
    }

    /// The entries and the index, mutably.
    fn parts_mut(&mut self) -> (&mut [Bucket<K, V>], &mut [u32]) {
        self.check();
        // SAFETY: as in `parts`; the two ranges are disjoint and `&mut self`
        // is unique.
        unsafe {
            let p = self.ptr.as_ptr();
            (
                core::slice::from_raw_parts_mut(p, self.len),
                core::slice::from_raw_parts_mut(p.add(usable(self.slots)).cast(), self.slots),
            )
        }
    }

    /// Sets the index size to `new_slots > self.slots`. Never moves the
    /// entries; rebuilds the index from the stored hashes. Runs no user
    /// code. On failure the map is unchanged.
    fn try_grow_to(&mut self, new_slots: usize) -> Result<(), AllocError> {
        debug_assert!(new_slots > self.slots);
        let st = self.frame.st;
        let bytes = block_bytes::<K, V>(new_slots)?;
        let old_end = self.end_off();
        // We hold `&mut self`, so by invariant I nothing above our end is
        // reachable; after this reset our block is the top block.
        st.reset_to(old_end);
        if self.allocated() {
            let start = st.offset_of(self.ptr.cast());
            let new_end = start.checked_add(bytes).ok_or(OVERFLOW)?;
            st.try_extend(old_end, new_end)?;
        } else {
            let start = st.try_bump(bytes, align_of::<Bucket<K, V>>())?;
            self.ptr = st.ptr_at(start).cast();
        }
        // SAFETY: the block now has `bytes` bytes: `usable(new_slots)`
        // entries, then room for `new_slots` slots, which hold no entry
        // because `usable(new_slots) > usable(self.slots) >= len`.
        unsafe {
            let index = self.ptr.as_ptr().add(usable(new_slots)).cast::<u32>();
            index.write_bytes(0, new_slots);
        }
        self.slots = new_slots;
        let (b, index) = self.parts_mut();
        fill_index(index, b.iter().map(|b| b.hash));
        Ok(())
    }

    /// The number of entries.
    pub fn len(&self) -> usize {
        self.len
    }

    /// Whether the map has no entries.
    pub fn is_empty(&self) -> bool {
        self.len == 0
    }

    /// The number of entries the map can hold without growing.
    pub fn capacity(&self) -> usize {
        usable(self.slots)
    }

    /// The map's hasher.
    pub fn hasher(&self) -> &S {
        &self.hasher
    }

    /// Makes room for at least `additional` more entries.
    ///
    /// # Panics
    /// If the stack is exhausted, the capacity overflows, or an entry needs
    /// alignment above 4096.
    pub fn reserve(&mut self, additional: usize) {
        self.try_reserve(additional).unwrap_or_else(|e| fail(e));
    }

    /// Makes room for at least `additional` more entries. On failure the
    /// map is unchanged (the stack may still have reclaimed unreachable space
    /// above it). Runs no user code.
    pub fn try_reserve(&mut self, additional: usize) -> Result<(), AllocError> {
        let needed = self.len.checked_add(additional).ok_or(OVERFLOW)?;
        if needed <= self.capacity() {
            return Ok(());
        }
        self.try_grow_to(slots_for(needed)?)
    }

    /// Appends an entry whose key is not in the map and returns its index.
    /// Runs no user code beyond dropping `key` and `value` if it panics.
    fn push(&mut self, hash: u64, key: K, value: V) -> usize {
        self.reserve(1);
        let len = self.len;
        self.check();
        // SAFETY: `len < capacity()` after `reserve`: slot `len` is inside the
        // block and holds no entry.
        unsafe {
            self.ptr
                .as_ptr()
                .add(len)
                .write(Bucket { hash, key, value })
        };
        self.len = len + 1;
        let (_, index) = self.parts_mut();
        insert_slot(index, hash, len);
        len
    }

    /// Removes the last entry, which no slot refers to any more.
    fn pop_bucket(&mut self) -> Bucket<K, V> {
        self.len -= 1;
        // SAFETY: entry `len` was initialised and is now outside `[0, len)`;
        // the caller removed the only slot that referred to it.
        unsafe { self.ptr.as_ptr().add(self.len).read() }
    }

    /// The entry at `index` in insertion order.
    pub fn get_index(&self, index: usize) -> Option<(&K, &V)> {
        self.parts().0.get(index).map(|b| (&b.key, &b.value))
    }

    /// The entry at `index`, with its value mutable.
    pub fn get_index_mut(&mut self, index: usize) -> Option<(&K, &mut V)> {
        self.parts_mut()
            .0
            .get_mut(index)
            .map(|b| (&b.key, &mut b.value))
    }

    /// Removes the entry at `index` and moves the last entry into its place.
    /// O(1). Runs no user code.
    pub fn swap_remove_index(&mut self, index: usize) -> Option<(K, V)> {
        if index >= self.len {
            return None;
        }
        let last = self.len - 1;
        let (b, ix) = self.parts_mut();
        let s = slot_of(ix, b[index].hash, index);
        remove_slot(ix, s, |e| b[e].hash);
        if index != last {
            let s = slot_of(ix, b[last].hash, last);
            ix[s] = index as u32 + 1;
            b.swap(index, last);
        }
        let b = self.pop_bucket();
        Some((b.key, b.value))
    }

    /// Removes the entry at `index` and shifts the later entries down,
    /// keeping their order. O(n). Runs no user code.
    pub fn shift_remove_index(&mut self, index: usize) -> Option<(K, V)> {
        if index >= self.len {
            return None;
        }
        let (b, ix) = self.parts_mut();
        let s = slot_of(ix, b[index].hash, index);
        remove_slot(ix, s, |e| b[e].hash);
        for v in ix.iter_mut() {
            if *v as usize > index + 1 {
                *v -= 1;
            }
        }
        b[index..].rotate_left(1);
        let b = self.pop_bucket();
        Some((b.key, b.value))
    }

    /// Iterates the entries in insertion order.
    pub fn iter(&self) -> Iter<'_, K, V> {
        Iter {
            inner: self.parts().0.iter(),
        }
    }

    /// Iterates the entries in insertion order, with mutable values.
    pub fn iter_mut(&mut self) -> IterMut<'_, K, V> {
        IterMut {
            inner: self.parts_mut().0.iter_mut(),
        }
    }

    /// Iterates the keys in insertion order.
    pub fn keys(&self) -> Keys<'_, K, V> {
        Keys { inner: self.iter() }
    }

    /// Iterates the values in insertion order.
    pub fn values(&self) -> Values<'_, K, V> {
        Values { inner: self.iter() }
    }

    /// Iterates the values in insertion order, mutably.
    pub fn values_mut(&mut self) -> ValuesMut<'_, K, V> {
        ValuesMut {
            inner: self.iter_mut(),
        }
    }

    /// Removes every entry, dropping them back to front. Keeps the
    /// capacity. If a destructor panics the remaining entries are still
    /// dropped and the map is left empty.
    pub fn clear(&mut self) {
        let len = core::mem::replace(&mut self.len, 0);
        let (_, index) = self.parts_mut();
        index.fill(0);
        // SAFETY: we own `len` initialised entries; `self.len` is now 0 and
        // the index is empty, so nothing refers to them.
        unsafe { drop_back_to_front(self.ptr.as_ptr(), len) };
    }

    /// Freeze and descend: a frame above this map. The map cannot be used
    /// until the frame and everything allocated through it are gone. The new
    /// frame starts at the end of the map's block, so the map keeps its spare
    /// room.
    ///
    /// # Examples
    /// ```
    /// use ordstack::{SMap, Stack};
    ///
    /// let mut st = Stack::with_reserve(1 << 16);
    /// let mut f = st.frame();
    /// let mut m: SMap<u32, u32> = SMap::new_in(&mut f);
    /// m.insert(1, 10);
    /// {
    ///     let mut up = m.above();
    ///     let b = up.alloc(2u32);
    ///     assert_eq!(*b, 2);
    /// }
    /// m.insert(2, 20);
    /// assert_eq!(m.len(), 2);
    /// ```
    pub fn above(&mut self) -> Frame<'_> {
        let end = self.end_off();
        self.frame.st.reset_to(end);
        Frame {
            st: self.frame.st,
            end,
        }
    }
}

impl<'v, 'f, K: Hash + Eq, V, S: BuildHasher> SMap<'v, 'f, K, V, S> {
    /// Inserts `value` under `key`. If the key was present, its value is
    /// replaced and returned, and the entry keeps its position (the old key
    /// is kept, `key` is dropped).
    ///
    /// # Panics
    /// If the stack is exhausted, the capacity overflows, or an entry needs
    /// alignment above 4096. If `Hash` or `Eq` panics, the map is unchanged.
    pub fn insert(&mut self, key: K, value: V) -> Option<V> {
        let hash = self.hasher.hash_one(&key);
        let (b, index) = self.parts();
        match lookup(b, index, hash, &key) {
            Some(i) => Some(core::mem::replace(&mut self.parts_mut().0[i].value, value)),
            None => {
                self.push(hash, key, value);
                None
            }
        }
    }

    /// The position of `key` in insertion order.
    pub fn get_index_of<Q>(&self, key: &Q) -> Option<usize>
    where
        K: Borrow<Q>,
        Q: ?Sized + Hash + Eq,
    {
        if self.len == 0 {
            return None;
        }
        let hash = self.hasher.hash_one(key);
        let (b, index) = self.parts();
        lookup(b, index, hash, key)
    }

    /// The value under `key`.
    pub fn get<Q>(&self, key: &Q) -> Option<&V>
    where
        K: Borrow<Q>,
        Q: ?Sized + Hash + Eq,
    {
        let i = self.get_index_of(key)?;
        Some(&self.parts().0[i].value)
    }

    /// The value under `key`, mutably.
    pub fn get_mut<Q>(&mut self, key: &Q) -> Option<&mut V>
    where
        K: Borrow<Q>,
        Q: ?Sized + Hash + Eq,
    {
        let i = self.get_index_of(key)?;
        Some(&mut self.parts_mut().0[i].value)
    }

    /// Whether `key` is in the map.
    pub fn contains_key<Q>(&self, key: &Q) -> bool
    where
        K: Borrow<Q>,
        Q: ?Sized + Hash + Eq,
    {
        self.get_index_of(key).is_some()
    }

    /// Removes `key` and returns its value; the last entry takes its place.
    /// O(1).
    pub fn swap_remove<Q>(&mut self, key: &Q) -> Option<V>
    where
        K: Borrow<Q>,
        Q: ?Sized + Hash + Eq,
    {
        let i = self.get_index_of(key)?;
        self.swap_remove_index(i).map(|(_, v)| v)
    }

    /// Removes `key` and returns its value, keeping the order of the other
    /// entries. O(n).
    pub fn shift_remove<Q>(&mut self, key: &Q) -> Option<V>
    where
        K: Borrow<Q>,
        Q: ?Sized + Hash + Eq,
    {
        let i = self.get_index_of(key)?;
        self.shift_remove_index(i).map(|(_, v)| v)
    }
}

impl<K, V, S> Drop for SMap<'_, '_, K, V, S> {
    fn drop(&mut self) {
        self.check();
        let len = core::mem::replace(&mut self.len, 0);
        // SAFETY: we own `len` initialised entries and `self.len` is now 0.
        // The block itself is reclaimed when the frame is next used.
        unsafe { drop_back_to_front(self.ptr.as_ptr(), len) };
    }
}

impl<K: Hash + Eq, V, S: BuildHasher> Extend<(K, V)> for SMap<'_, '_, K, V, S> {
    fn extend<I: IntoIterator<Item = (K, V)>>(&mut self, iter: I) {
        let iter = iter.into_iter();
        // A hint that does not fit is not an error; `insert` decides.
        let _ = self.try_reserve(iter.size_hint().0);
        for (k, v) in iter {
            self.insert(k, v);
        }
    }
}

impl<'v, 'f, K: Hash + Eq, V, S: BuildHasher> SMap<'v, 'f, K, V, S> {
    /// The entry for `key`, for in-place insertion or update.
    pub fn entry(&mut self, key: K) -> Entry<'_, 'v, 'f, K, V, S> {
        let hash = self.hasher.hash_one(&key);
        let (b, index) = self.parts();
        match lookup(b, index, hash, &key) {
            Some(index) => Entry::Occupied(OccupiedEntry { map: self, index }),
            None => Entry::Vacant(VacantEntry {
                map: self,
                hash,
                key,
            }),
        }
    }
}

impl<'a, K, V, S> Entry<'a, '_, '_, K, V, S> {
    /// The entry's key.
    pub fn key(&self) -> &K {
        match self {
            Entry::Occupied(e) => e.key(),
            Entry::Vacant(e) => e.key(),
        }
    }

    /// The value, inserting `default` if the key is absent.
    ///
    /// # Panics
    /// As [`SMap::insert`].
    pub fn or_insert(self, default: V) -> &'a mut V {
        match self {
            Entry::Occupied(e) => e.into_mut(),
            Entry::Vacant(e) => e.insert(default),
        }
    }

    /// The value, inserting `f()` if the key is absent. `f` runs before
    /// anything is reserved.
    ///
    /// # Panics
    /// As [`SMap::insert`].
    pub fn or_insert_with(self, f: impl FnOnce() -> V) -> &'a mut V {
        match self {
            Entry::Occupied(e) => e.into_mut(),
            Entry::Vacant(e) => e.insert(f()),
        }
    }

    /// The value, inserting `V::default()` if the key is absent.
    ///
    /// # Panics
    /// As [`SMap::insert`].
    pub fn or_default(self) -> &'a mut V
    where
        V: Default,
    {
        self.or_insert_with(V::default)
    }
}

impl<'a, K, V, S> OccupiedEntry<'a, '_, '_, K, V, S> {
    /// The key in the map.
    pub fn key(&self) -> &K {
        &self.map.parts().0[self.index].key
    }

    /// The entry's position in insertion order.
    pub fn index(&self) -> usize {
        self.index
    }

    /// The value.
    pub fn get(&self) -> &V {
        &self.map.parts().0[self.index].value
    }

    /// The value, mutably.
    pub fn get_mut(&mut self) -> &mut V {
        &mut self.map.parts_mut().0[self.index].value
    }

    /// The value, mutably, for as long as the map was borrowed.
    pub fn into_mut(self) -> &'a mut V {
        &mut self.map.parts_mut().0[self.index].value
    }
}

impl<'a, K, V, S> VacantEntry<'a, '_, '_, K, V, S> {
    /// The key that would be inserted.
    pub fn key(&self) -> &K {
        &self.key
    }

    /// Inserts the key with `value` at the end of the map.
    ///
    /// # Panics
    /// If the stack is exhausted, the capacity overflows, or an entry needs
    /// alignment above 4096.
    pub fn insert(self, value: V) -> &'a mut V {
        let i = self.map.push(self.hash, self.key, value);
        &mut self.map.parts_mut().0[i].value
    }
}

```rust
impl<'v, 'f, K, V, S> SMap<'v, 'f, K, V, S> {
    fn into_parts(self) -> (&'v mut Frame<'f>, NonNull<Bucket<K, V>>, usize, usize, S) {
        let this = ManuallyDrop::new(self);
        // SAFETY: `this` is never used or dropped again, so this moves the
        // unique frame reference and the hasher out of it.
        let (frame, hasher) =
            unsafe { (core::ptr::read(&this.frame), core::ptr::read(&this.hasher)) };
        (frame, this.ptr, this.len, this.slots, hasher)
    }

    /// Freezes the map in place, without moving the entries. The index is
    /// rebuilt right after the entries at the size `len` entries need, the
    /// rest of the block goes back to the stack, and the frame can be used
    /// again. Runs no user code.
    ///
    /// # Examples
    /// ```
    /// use ordstack::{SMap, Stack};
    ///
    /// let mut st = Stack::with_reserve(1 << 16);
    /// let mut f = st.frame();
    /// let mut m: SMap<u32, &str> = SMap::new_in(&mut f);
    /// m.insert(7, "seven");
    /// let frozen = m.freeze();
    /// let b = f.alloc(1u8); // sits beside `frozen`
    /// assert_eq!((frozen.get(&7), *b), (Some(&"seven"), 1));
    /// ```
    pub fn freeze(self) -> SFrozenMap<'f, K, V, S> {
        self.check();
        let (frame, ptr, len, slots, hasher) = self.into_parts();
        if slots == 0 {
            return SFrozenMap::empty(hasher);
        }
        let st = frame.st;
        if len == 0 {
            st.reset_to(frame.end);
            return SFrozenMap::empty(hasher);
        }
        // `len <= usable(slots)`, so the compact index is no larger.
        let new_slots = slots_for(len).unwrap_or_else(|e| fail(e));
        let new_end = st.offset_of(ptr.cast()) + len * size_of::<Bucket<K, V>>() + new_slots * 4;
        // SAFETY: `[len, len + new_slots * 4 bytes)` past the entries lies
        // inside the old block (the old block ends at or after `new_end`),
        // and holds no entry.
        unsafe {
            ptr.as_ptr()
                .add(len)
                .cast::<u32>()
                .write_bytes(0, new_slots)
        };
        st.reset_to(new_end);
        frame.end = new_end;
        // SAFETY: `len` initialised entries followed by `new_slots` zeroed
        // slots, all below `frame.end`, which does not decrease while `'f` is
        // alive.
        let mut m = unsafe { SFrozenMap::from_raw_parts(ptr, len, new_slots, hasher) };
        let (b, index) = m.parts_mut();
        fill_index(index, b.iter().map(|b| b.hash));
        m
    }

    /// Moves the entries into a frozen block of `dst` with one bitwise
    /// copy, and builds the index there. This map's block is left to be
    /// reclaimed. As with [`SVec::freeze_into`](crate::SVec::freeze_into),
    /// `dst` is in practice a frame of another stack. Runs no user code.
    ///
    /// # Panics
    /// If `dst`'s stack is exhausted, or an entry needs alignment above
    /// 4096. The map is then dropped with its entries during unwinding.
    pub fn freeze_into<'g>(mut self, dst: &mut Frame<'g>) -> SFrozenMap<'g, K, V, S> {
        self.check();
        let len = self.len;
        if len == 0 {
            let (_, _, _, _, hasher) = self.into_parts();
            return SFrozenMap::empty(hasher);
        }
        let new_slots = slots_for(len).unwrap_or_else(|e| fail(e));
        let units = len + (new_slots * 4).div_ceil(size_of::<Bucket<K, V>>());
        let p = dst
            .try_alloc_array::<Bucket<K, V>>(units)
            .unwrap_or_else(|e| fail(e));
        // SAFETY: `p` is a fresh block of `units` entries' size, which holds
        // `len` entries and then `new_slots` slots, and cannot overlap our
        // block. Setting `self.len = 0` hands the entries to the copy.
        unsafe {
            core::ptr::copy_nonoverlapping(self.ptr.as_ptr(), p.as_ptr(), len);
            self.len = 0;
            p.as_ptr().add(len).cast::<u32>().write_bytes(0, new_slots);
        }
        let (_, _, _, _, hasher) = self.into_parts();
        // SAFETY: `len` initialised entries followed by `new_slots` zeroed
        // slots, in a block `dst` allocated for `'g`.
        let mut m = unsafe { SFrozenMap::from_raw_parts(p, len, new_slots, hasher) };
        let (b, index) = m.parts_mut();
        fill_index(index, b.iter().map(|b| b.hash));
        m
    }
}

/// A frozen, insertion-ordered hash map: a frozen block holding the entries
/// and a compact index. Lookups and iteration as for [`SMap`]; no
/// insertion or removal. Entries are dropped back to front.
pub struct SFrozenMap<'f, K, V, S = RandomState> {
    ptr: NonNull<Bucket<K, V>>,
    len: usize,
    /// Index slots, right after the `len` entries; 0 when empty.
    slots: usize,
    hasher: S,
    _brand: PhantomData<(&'f (), Bucket<K, V>)>,
}

impl<'f, K, V, S> SFrozenMap<'f, K, V, S> {
    fn empty(hasher: S) -> Self {
        SFrozenMap {
            ptr: NonNull::dangling(),
            len: 0,
            slots: 0,
            hasher,
            _brand: PhantomData,
        }
    }

    /// # Safety
    /// `ptr` must point to `len` initialised entries followed by `slots`
    /// initialised slots, in a block that stays allocated and unaliased for
    /// `'f`.
    unsafe fn from_raw_parts(
        ptr: NonNull<Bucket<K, V>>,
        len: usize,
        slots: usize,
        hasher: S,
    ) -> Self {
        SFrozenMap {
            ptr,
            len,
            slots,
            hasher,
            _brand: PhantomData,
        }
    }

    fn check(&self) {
        debug::check_live(
            self.ptr.as_ptr().cast(),
            self.len * size_of::<Bucket<K, V>>() + self.slots * 4,
        );
    }

    fn parts(&self) -> (&[Bucket<K, V>], &[u32]) {
        self.check();
        // SAFETY: handle contract (`from_raw_parts`): `len` entries, then
        // `slots` slots, disjoint, alive for the brand. When empty both
        // lengths are 0 and the pointer is dangling and aligned.
        unsafe {
            let p = self.ptr.as_ptr();
            (
                core::slice::from_raw_parts(p, self.len),
                core::slice::from_raw_parts(p.add(self.len).cast(), self.slots),
            )
        }
    }

    fn parts_mut(&mut self) -> (&mut [Bucket<K, V>], &mut [u32]) {
        self.check();
        // SAFETY: as in `parts`; `&mut self` is unique.
        unsafe {
            let p = self.ptr.as_ptr();
            (
                core::slice::from_raw_parts_mut(p, self.len),
                core::slice::from_raw_parts_mut(p.add(self.len).cast(), self.slots),
            )
        }
    }

    /// The number of entries.
    pub fn len(&self) -> usize {
        self.len
    }

    /// Whether the map has no entries.
    pub fn is_empty(&self) -> bool {
        self.len == 0
    }

    /// The map's hasher.
    pub fn hasher(&self) -> &S {
        &self.hasher
    }

    /// The entry at `index` in insertion order.
    pub fn get_index(&self, index: usize) -> Option<(&K, &V)> {
        self.parts().0.get(index).map(|b| (&b.key, &b.value))
    }

    /// The entry at `index`, with its value mutable.
    pub fn get_index_mut(&mut self, index: usize) -> Option<(&K, &mut V)> {
        self.parts_mut()
            .0
            .get_mut(index)
            .map(|b| (&b.key, &mut b.value))
    }

    /// Iterates the entries in insertion order.
    pub fn iter(&self) -> Iter<'_, K, V> {
        Iter {
            inner: self.parts().0.iter(),
        }
    }

    /// Iterates the entries in insertion order, with mutable values.
    pub fn iter_mut(&mut self) -> IterMut<'_, K, V> {
        IterMut {
            inner: self.parts_mut().0.iter_mut(),
        }
    }

    /// Iterates the keys in insertion order.
    pub fn keys(&self) -> Keys<'_, K, V> {
        Keys { inner: self.iter() }
    }

    /// Iterates the values in insertion order.
    pub fn values(&self) -> Values<'_, K, V> {
        Values { inner: self.iter() }
    }

    /// Iterates the values in insertion order, mutably.
    pub fn values_mut(&mut self) -> ValuesMut<'_, K, V> {
        ValuesMut {
            inner: self.iter_mut(),
        }
    }
}

impl<K: Hash + Eq, V, S: BuildHasher> SFrozenMap<'_, K, V, S> {
    /// The position of `key` in insertion order.
    pub fn get_index_of<Q>(&self, key: &Q) -> Option<usize>
    where
        K: Borrow<Q>,
        Q: ?Sized + Hash + Eq,
    {
        if self.len == 0 {
            return None;
        }
        let hash = self.hasher.hash_one(key);
        let (b, index) = self.parts();
        lookup(b, index, hash, key)
    }

    /// The value under `key`.
    pub fn get<Q>(&self, key: &Q) -> Option<&V>
    where
        K: Borrow<Q>,
        Q: ?Sized + Hash + Eq,
    {
        let i = self.get_index_of(key)?;
        Some(&self.parts().0[i].value)
    }

    /// The value under `key`, mutably.
    pub fn get_mut<Q>(&mut self, key: &Q) -> Option<&mut V>
    where
        K: Borrow<Q>,
        Q: ?Sized + Hash + Eq,
    {
        let i = self.get_index_of(key)?;
        Some(&mut self.parts_mut().0[i].value)
    }

    /// Whether `key` is in the map.
    pub fn contains_key<Q>(&self, key: &Q) -> bool
    where
        K: Borrow<Q>,
        Q: ?Sized + Hash + Eq,
    {
        self.get_index_of(key).is_some()
    }
}

impl<K, V, S> Drop for SFrozenMap<'_, K, V, S> {
    fn drop(&mut self) {
        self.check();

/// Runs operations on an `SMap<u8, Tracked>` on top of `up`, against a
/// model of its entries in insertion order. Keys are `arg % 32`, so they
/// collide often. Decoding (`byte % 8`; `arg` as in `level`):
///
/// | op   | operation                                                        |
/// |------|------------------------------------------------------------------|
/// | 0..=2 | insert a fresh value under a key (replacing drops the old one)  |
/// | 3    | `swap_remove` a key                                              |
/// | 4    | `shift_remove` a key                                             |
/// | 5    | `entry(key).or_insert_with(fresh)`                               |
/// | 6    | allocate a box above the map, check it, drop it                  |
/// | 7    | end the map: by `arg % 3`, drop it; `freeze` it and allocate a box beside it; or `freeze_into` the scratch frame |
///
/// A script that runs out ends the map by dropping it.
fn map_scope<'c>(
    up: &mut Frame<'_>,
    scratch: &mut Frame<'_>,
    input: &mut dyn Iterator<Item = u8>,
    cx: &'c Cx,
) {
    let mut map: SMap<'_, '_, u8, Tracked<'c>> = SMap::new_in(up);
    let mut m: Vec<(u8, u32)> = Vec::new();
    let mut end = 0;
    let key = |input: &mut dyn Iterator<Item = u8>| (arg(input) % 32) as u8;
    while let Some(op) = input.next() {
        match op % 8 {
            0..=2 => {
                let k = key(input);
                let t = cx.make();
                let id = t.id;
                match m.iter_mut().find(|e| e.0 == k) {
                    Some(e) => {
                        let old = map.insert(k, t).expect("the model has this key");
                        assert_eq!(old.id, std::mem::replace(&mut e.1, id));
                        cx.expect([old.id]);
                    }
                    None => {
                        assert!(map.insert(k, t).is_none());
                        m.push((k, id));
                    }
                }
            }
            3 => {
                let k = key(input);
                let want = m.iter().position(|e| e.0 == k).map(|p| m.swap_remove(p).1);
                let got = map.swap_remove(&k);
                assert_eq!(got.as_ref().map(|t| t.id), want);
                cx.expect(want);
            }
            4 => {
                let k = key(input);
                let want = m.iter().position(|e| e.0 == k).map(|p| m.remove(p).1);
                let got = map.shift_remove(&k);
                assert_eq!(got.as_ref().map(|t| t.id), want);
                cx.expect(want);
            }
            5 => {
                let k = key(input);
                let before = cx.next.get();
                let id = map.entry(k).or_insert_with(|| cx.make()).id;
                match m.iter().find(|e| e.0 == k) {
                    Some(e) => {
                        assert_eq!(e.1, id);
                        assert_eq!(
                            cx.next.get(),
                            before,
                            "or_insert_with ran for a present key"
                        );
                    }
                    None => m.push((k, id)),
                }
            }
            6 => {
                let t = cx.make();
                let id = t.id;
                let b = map.above().alloc(t);
                assert_eq!(b.id, id);
                cx.expect([id]);
                drop(b);
            }
            _ => {
                end = arg(input) % 3;
                break;
            }
        }
        check_map(
            map.iter().map(|(k, t)| (*k, t.id)).collect(),
            |k| map.get(&k).map(|t| t.id),
            &m,
        );
    }
    match end {
        0 => {
            cx.expect(m.iter().rev().map(|e| e.1));
            drop(map);
        }
        1 => {
            let fm = map.freeze();
            let t = cx.make();
            let id = t.id;
            let b = up.alloc(t);
            check_map(
                fm.iter().map(|(k, t)| (*k, t.id)).collect(),

            );
            cx.expect(m.iter().rev().map(|e| e.1));
            drop(fm);
        }
    }
